## EA-FT EEGNet Integration

This section extends the notebook from dataset exploration to the proposed **EA-FT EEGNet** pipeline used in the proposal:

1. Load Graz 2a trials.
2. Keep only **left-hand vs right-hand** motor imagery.
3. Apply **bandpass filtering** in the motor-imagery rhythm range.
4. Apply **Euclidean Alignment (EA)** to reduce subject/session distribution shift.
5. Train an EEGNet source model.
6. Fine-tune only the higher/adaptation layers using a small amount of target-subject calibration data.

For today’s research meeting, you can explain this as: **EA aligns the input space; fine-tuning adapts the model weights using only a small amount of target-user data.**

### 1. Trial extraction for left vs right hand

BCI Competition IV 2a uses labels:

- `1` = left hand
- `2` = right hand

For this research scope, we keep only labels `1` and `2`, then remap them to:

- `0` = left hand
- `1` = right hand

The model input format will be **trials × channels × time**.

===== CELL 25 [code] =====
def bandpass_filter_trials(trials, fs, low=8.0, high=30.0, order=4):
    """Bandpass filter trials shaped (n_trials, n_channels, n_times)."""
    nyq = 0.5 * fs
    b, a = butter(order, [low / nyq, high / nyq], btype="band")
    return filtfilt(b, a, trials, axis=-1)


def extract_left_right_trials_from_mat(path: Path, tmin=0.5, tmax=4.0, low=8.0, high=30.0):
    """
    Load a Graz 2a .mat file and return left/right trials.

    Returns
    -------
    X_trials : np.ndarray
        Shape: (n_trials, n_channels, n_times)
    y_trials : np.ndarray
        Shape: (n_trials,), where 0=left and 1=right
    fs : int
        Sampling rate
    """
    d = loadmat(path)
    block = d["data"]
    X_all, y_all = [], []
    fs_ref = None

    for session_idx in range(block.shape[1]):
        session = block[0, session_idx]
        X_cont = session["X"][0, 0]          # time x channels
        y = session["y"][0, 0].squeeze()
        trial = session["trial"][0, 0].squeeze()
        fs = int(session["fs"][0, 0].squeeze())
        fs_ref = fs

        if y.size == 0 or trial.size == 0:
            continue

        start_offset = int(tmin * fs)
        stop_offset = int(tmax * fs)

        for onset, label in zip(trial, y):
            label = int(label)
            if label not in [1, 2]:
                continue

            start = int(onset) + start_offset
            stop = int(onset) + stop_offset

            if start >= 0 and stop <= X_cont.shape[0]:
                segment = X_cont[start:stop, :].T  # channels x time
                X_all.append(segment)
                y_all.append(0 if label == 1 else 1)

    if not X_all:
        raise ValueError(f"No left/right labelled trials found in {path.name}")

    X_trials = np.stack(X_all).astype(np.float32)
    y_trials = np.array(y_all, dtype=np.int64)

    X_trials = bandpass_filter_trials(X_trials, fs_ref, low=low, high=high).astype(np.float32)
    return X_trials, y_trials, fs_ref

### 2. Euclidean Alignment (EA)

EA is an unsupervised input-space alignment method. For each subject/session, we estimate the average covariance matrix across trials and whiten the trials using its inverse square root.

Mathematically:

\[
\bar{R}=\frac{1}{N}\sum_{i=1}^{N} X_iX_i^T
\]

\[
\tilde{X_i}=\bar{R}^{-1/2}X_i
\]

After EA, the average covariance becomes closer to the identity matrix. This helps reduce subject-to-subject distribution differences before EEGNet receives the signal.

===== CELL 27 [code] =====
def fit_euclidean_alignment(X, eps=1e-6):
    """
    Fit Euclidean Alignment reference matrix.

    Parameters
    ----------
    X : np.ndarray
        Trials shaped (n_trials, n_channels, n_times).
    eps : float
        Small diagonal regularization for numerical stability.

    Returns
    -------
    R_inv_sqrt : np.ndarray
        Inverse square root of average covariance, shaped (n_channels, n_channels).
    R_bar : np.ndarray
        Average covariance matrix before whitening.
    """
    n_trials, n_channels, _ = X.shape
    covs = np.zeros((n_trials, n_channels, n_channels), dtype=np.float64)

    for i in range(n_trials):
        Xi = X[i]
        covs[i] = (Xi @ Xi.T) / Xi.shape[1]

    R_bar = covs.mean(axis=0)
    R_bar += eps * np.eye(n_channels)
    R_inv_sqrt = fractional_matrix_power(R_bar, -0.5).real
    return R_inv_sqrt.astype(np.float32), R_bar.astype(np.float32)


def apply_euclidean_alignment(X, R_inv_sqrt):
    """Apply fitted EA matrix to trials shaped (n_trials, n_channels, n_times)."""
    return np.asarray([R_inv_sqrt @ Xi for Xi in X], dtype=np.float32)


def euclidean_align_subject(X):
    """Convenience function: fit EA on a subject/session and transform that same data."""
    R_inv_sqrt, R_bar = fit_euclidean_alignment(X)
    X_ea = apply_euclidean_alignment(X, R_inv_sqrt)
    return X_ea, R_inv_sqrt, R_bar

def make_transfer_split(subjects, target_subject, calibration_trials_per_class=10, test_size=0.5):
    """
    Create source training data and target calibration/test data.

    calibration_trials_per_class controls the simulated target-user calibration amount.
    Example: 10 trials/class is a small calibration set compared with full subject-specific training.
    """
    source_X, source_y = [], []

    for subj, item in subjects.items():
        if subj == target_subject:
            continue
        source_X.append(item["X_ea"])
        source_y.append(item["y"])

    if not source_X:
        raise ValueError("Need at least one source subject and one target subject.")

    X_source = np.concatenate(source_X, axis=0)
    y_source = np.concatenate(source_y, axis=0)

    X_target = subjects[target_subject]["X_ea"]
    y_target = subjects[target_subject]["y"]

    # First create a candidate calibration/test split with stratification.
    X_cal_pool, X_test, y_cal_pool, y_test = train_test_split(
        X_target,
        y_target,
        test_size=test_size,
        stratify=y_target,
        random_state=RANDOM_STATE,
    )

    # Select a small balanced calibration subset.
    chosen_idx = []
    for cls in [0, 1]:
        idx = np.where(y_cal_pool == cls)[0]
        n_take = min(calibration_trials_per_class, len(idx))
        chosen_idx.extend(idx[:n_take])

    chosen_idx = np.array(chosen_idx)
    X_cal = X_cal_pool[chosen_idx]
    y_cal = y_cal_pool[chosen_idx]

    return X_source, y_source, X_cal, y_cal, X_test, y_test

### 4. Compact EEGNet model

This EEGNet implementation follows the compact CNN idea used for EEG-based BCI classification:

- temporal convolution learns frequency-related filters,
- depthwise spatial convolution learns channel-specific spatial filters,
- separable convolution reduces parameters,
- final classifier predicts left vs right hand.

For fine-tuning, we will freeze the lower feature extraction blocks and update only the later adaptation/classification layers.

===== CELL 33 [code] =====
if TORCH_AVAILABLE:
    class EEGNet(nn.Module):
        def __init__(self, n_channels, n_times, n_classes=2, F1=8, D=2, F2=16, dropout=0.5):
            super().__init__()
            self.block1 = nn.Sequential(
                nn.Conv2d(1, F1, kernel_size=(1, 64), padding=(0, 32), bias=False),
                nn.BatchNorm2d(F1),
                nn.Conv2d(F1, F1 * D, kernel_size=(n_channels, 1), groups=F1, bias=False),
                nn.BatchNorm2d(F1 * D),
                nn.ELU(),
                nn.AvgPool2d(kernel_size=(1, 4)),
                nn.Dropout(dropout),
            )
            self.block2 = nn.Sequential(
                nn.Conv2d(F1 * D, F1 * D, kernel_size=(1, 16), padding=(0, 8), groups=F1 * D, bias=False),
                nn.Conv2d(F1 * D, F2, kernel_size=(1, 1), bias=False),
                nn.BatchNorm2d(F2),
                nn.ELU(),
                nn.AvgPool2d(kernel_size=(1, 8)),
                nn.Dropout(dropout),
            )

            with torch.no_grad():
                dummy = torch.zeros(1, 1, n_channels, n_times)
                out = self.block2(self.block1(dummy))
                self.flattened_features = out.numel()

            self.classifier = nn.Linear(self.flattened_features, n_classes)

        def forward(self, x):
            # x shape: batch x channels x time
            x = x.unsqueeze(1)  # batch x 1 x channels x time
            x = self.block1(x)
            x = self.block2(x)
            x = x.flatten(start_dim=1)
            return self.classifier(x)

def make_loader(X, y, batch_size=32, shuffle=True):
    X_t = torch.tensor(X, dtype=torch.float32)
    y_t = torch.tensor(y, dtype=torch.long)
    return DataLoader(TensorDataset(X_t, y_t), batch_size=batch_size, shuffle=shuffle)


def train_one_phase(model, train_loader, val_loader=None, epochs=20, lr=1e-3):
    criterion = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(filter(lambda p: p.requires_grad, model.parameters()), lr=lr)

    history = {"train_loss": [], "val_acc": []}

    for epoch in range(1, epochs + 1):
        model.train()
        losses = []

        for xb, yb in train_loader:
            xb, yb = xb.to(DEVICE), yb.to(DEVICE)
            optimizer.zero_grad()
            logits = model(xb)
            loss = criterion(logits, yb)
            loss.backward()
            optimizer.step()
            losses.append(loss.item())

        mean_loss = float(np.mean(losses))
        history["train_loss"].append(mean_loss)

        if val_loader is not None:
            val_acc = evaluate_accuracy(model, val_loader)
            history["val_acc"].append(val_acc)
            print(f"Epoch {epoch:02d}/{epochs} | loss={mean_loss:.4f} | val_acc={val_acc:.3f}")
        else:
            print(f"Epoch {epoch:02d}/{epochs} | loss={mean_loss:.4f}")

    return history


def evaluate_accuracy(model, loader):
    model.eval()
    y_true, y_pred = [], []
    with torch.no_grad():
        for xb, yb in loader:
            xb = xb.to(DEVICE)
            logits = model(xb)
            pred = torch.argmax(logits, dim=1).cpu().numpy()
            y_pred.extend(pred.tolist())
            y_true.extend(yb.numpy().tolist())
    return accuracy_score(y_true, y_pred)


def predict_labels(model, loader):
    model.eval()
    y_true, y_pred = [], []
    with torch.no_grad():
        for xb, yb in loader:
            xb = xb.to(DEVICE)
            logits = model(xb)
            pred = torch.argmax(logits, dim=1).cpu().numpy()
            y_pred.extend(pred.tolist())
            y_true.extend(yb.numpy().tolist())
    return np.array(y_true), np.array(y_pred)

### 6. Fine-tuning integration (EA-FT)

Now we apply the proposed **EA-FT** idea:

- use the EA-aligned target calibration data,
- freeze earlier EEGNet layers to preserve general MI rhythm features,
- fine-tune only later/adaptation layers and classifier,
- evaluate on unseen target-subject trials.

This matches the proposal’s idea: **input-level alignment + conservative weight-level adaptation**.

===== CELL 38 [code] =====
def freeze_for_ea_ft(model, train_block2=True, train_classifier=True):
    """
    Freeze Block 1 and fine-tune Block 2 + classifier.
    This represents conservative layer-wise fine-tuning.
    """
    for param in model.parameters():
        param.requires_grad = False

    if train_block2:
        for param in model.block2.parameters():
            param.requires_grad = True

    if train_classifier:
        for param in model.classifier.parameters():
            param.requires_grad = True

    trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
    total = sum(p.numel() for p in model.parameters())
    print(f"Trainable parameters after freezing: {trainable}/{total}")
    return model
